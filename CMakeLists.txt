cmake_minimum_required(VERSION 3.16)
project(125A_EBM_Prototype VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ebm_core STATIC src/Engine.cpp src/Composer.cpp src/Novelty.cpp src/Session.cpp src/MidiWriter.cpp)
target_include_directories(ebm_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
    target_compile_options(ebm_core PRIVATE /W4 /WX)
else()
    target_compile_options(ebm_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
