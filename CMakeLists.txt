cmake_minimum_required(VERSION 3.16)
project(125A_EBM_Prototype VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ebm_core STATIC src/Engine.cpp src/MidiWriter.cpp)
target_include_directories(ebm_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
    target_compile_options(ebm_core PRIVATE /W4 /WX)
else()
    target_compile_options(ebm_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(ebm_batch tools/ebm_batch.cpp)
target_link_libraries(ebm_batch PRIVATE ebm_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(ebm_tests tests/engine_tests.cpp)
    target_link_libraries(ebm_tests PRIVATE ebm_core)
    add_test(NAME EBM_Engine_Contract COMMAND ebm_tests)
endif()
